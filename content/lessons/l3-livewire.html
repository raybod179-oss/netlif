Livewire چیست؟

Livewire اجازه می‌دهد صفحه‌های تعاملی (جست‌وجوی زنده، سبد خرید، فرم‌های پویا) را فقط با PHP و Blade بسازی، بدون اینکه خودت جاوااسکریپت و API بنویسی. کامپوننت در سرور اجرا می‌شود و تغییرات کوچک HTML به مرورگر می‌رسد.

انتخابکی مناسب است؟
Blade سادهصفحه‌های ثابت و فرم‌های عادی
Livewireتعامل زیاد بدون ساخت فرانت جدا؛ پنل ادمین، جست‌وجو، سبد
API + React/Vueاپ موبایل هم داری یا فرانت‌اند مستقل می‌خواهی
کدbash
composer require livewire/livewire
php artisan make:livewire ProductSearch
نسخه

این درس بر اساس Livewire 3 است. مفاهیم در نسخه‌های جدیدتر هم همین‌اند؛ اگر نام یک ویژگی فرق داشت، مستندات رسمی را ببین (لینک انتهای درس).

اولین کامپوننت: جست‌وجوی زنده

کدphp
<?php

namespace App\Livewire;

use App\Models\Product;
use Livewire\Attributes\Url;
use Livewire\Component;
use Livewire\WithPagination;

class ProductSearch extends Component
{
    use WithPagination;

    #[Url]                       // مقدار در آدرس هم می‌آید: /products?search=موس
    public string $search = '';

    public function updatingSearch(): void
    {
        $this->resetPage();      // با تغییر جست‌وجو برگرد صفحه‌ی ۱
    }

    public function render()
    {
        $products = Product::query()
            ->when($this->search, fn ($q, $term) => $q->where('title', 'like', "%{$term}%"))
            ->latest()
            ->paginate(12);

        return view('livewire.product-search', ['products' => $products]);
    }
}
کدhtml
<div>
    <input type="search" wire:model.live.debounce.300ms="search" placeholder="جست‌وجوی محصول...">

    <div wire:loading>در حال جست‌وجو...</div>

    <ul>
        @foreach ($products as $product)
            <li wire:key="product-{{ $product->id }}">{{ $product->title }}</li>
        @endforeach
    </ul>

    {{ $products->links() }}
</div>
کدphp
<?php

// routes/web.php  — کامپوننت تمام‌صفحه
use App\Livewire\ProductSearch;
use Illuminate\Support\Facades\Route;

Route::get('/products', ProductSearch::class);
  • wire:model.live تغییر را همان لحظه به سرور می‌فرستد؛ .debounce.300ms تا ۰٫۳ ثانیه بعد از آخرین کلید صبر می‌کند (جلوگیری از درخواست زیاد). بدون .live مقدار فقط با اولین اکشن (مثل ارسال فرم) می‌رود.
  • wire:key داخل حلقه لازم است تا Livewire هر ردیف را درست دنبال کند.
  • کامپوننت باید یک المنت ریشه داشته باشد.

اکشن، اعتبارسنجی و رویداد: سبد خرید

کدphp
<?php

namespace App\Livewire;

use App\Models\Product;
use Livewire\Attributes\Locked;
use Livewire\Attributes\On;
use Livewire\Attributes\Validate;
use Livewire\Component;

class AddToCart extends Component
{
    #[Locked]                              // کلاینت نمی‌تواند مقدارش را دستکاری کند
    public int $productId;

    #[Validate('required|integer|min:1|max:10')]
    public int $qty = 1;

    public function mount(Product $product): void
    {
        $this->productId = $product->id;
    }

    public function add(): void
    {
        $this->validate();

        $product = Product::findOrFail($this->productId);   // همیشه از دیتابیس بخوان
        abort_if($product->stock < $this->qty, 422, 'موجودی کافی نیست');

        $cart = session('cart', []);
        $cart[$product->id] = ($cart[$product->id] ?? 0) + $this->qty;
        session(['cart' => $cart]);

        $this->dispatch('cart-updated');                    // به کامپوننت‌های دیگر خبر بده
    }

    public function render()
    {
        return view('livewire.add-to-cart');
    }
}
کدhtml
<div>
    <input type="number" wire:model="qty" min="1" max="10">
    @error('qty') <small>{{ $message }}</small> @enderror

    <button wire:click="add" wire:loading.attr="disabled">افزودن به سبد</button>
</div>
کدphp
<?php

namespace App\Livewire;

use Livewire\Attributes\On;
use Livewire\Component;

class CartBadge extends Component
{
    public int $count = 0;

    public function mount(): void
    {
        $this->refreshCount();
    }

    #[On('cart-updated')]                  // وقتی AddToCart رویداد را فرستاد
    public function refreshCount(): void
    {
        $this->count = array_sum(session('cart', []));
    }

    public function render()
    {
        return view('livewire.cart-badge');
    }
}

امنیت Livewire؛ مهم‌ترین بخش

هر چیزی که public است دست کاربر است

ویژگی‌های public کامپوننت را مرورگر می‌تواند دستکاری کند و متدهای public را مستقیم صدا بزند. پس مثل یک API رفتار کن:

  • شناسه‌های حساس را با #[Locked] قفل کن یا اصلاً public نکن.
  • داخل هر اکشن دوباره authorize کن (Gate::authorize('update', $order))؛ فرض نکن چون دکمه مخفی بود، کسی صدایش نمی‌زند.
  • مقدارها را همیشه validate کن و قیمت/موجودی را از دیتابیس بخوان نه از ویژگی کامپوننت.
  • ویژگی‌های حجیم (مثل کل کالکشن‌ها) را public نگه نداری؛ حجم درخواست و نشت داده را بالا می‌برد.
کدphp
<?php

namespace App\Livewire;

use App\Models\Order;
use Illuminate\Support\Facades\Gate;
use Livewire\Attributes\Locked;
use Livewire\Component;

class OrderNote extends Component
{
    #[Locked]
    public int $orderId;

    public string $note = '';

    public function save(): void
    {
        $order = Order::findOrFail($this->orderId);
        Gate::authorize('update', $order);            // ✅ حتی داخل Livewire

        $this->validate(['note' => 'nullable|string|max:500']);
        $order->update(['note' => $this->note]);
    }

    public function render()
    {
        return view('livewire.order-note');
    }
}

فرم‌های تمیز: Form Object، اعتبارسنجی لحظه‌ای و آپلود فایل

وقتی فرم بزرگ شد، همه‌ی ویژگی‌ها و قوانین را در خود کامپوننت نگذار؛ یک «Form Object» بساز. (این درس بر پایه‌ی Livewire 3 است.)

کدbash
php artisan livewire:form ProductForm
کدphp
// app/Livewire/Forms/ProductForm.php
namespace App\Livewire\Forms;

use App\Models\Product;
use Livewire\Attributes\Validate;
use Livewire\Form;

class ProductForm extends Form
{
    #[Validate('required|string|min:3|max:120')]
    public string $name = '';

    #[Validate('required|integer|min:1000')]
    public string|int $price = '';

    #[Validate('nullable|image|max:2048')]   // حداکثر ۲ مگابایت
    public $image;

    public function store(): Product
    {
        $this->validate();

        return Product::create([
            'name'  => $this->name,
            'price' => (int) $this->price,
            'image' => $this->image?->store('products', 'public'),
        ]);
    }
}
کدphp
// app/Livewire/ProductCreate.php
use App\Livewire\Forms\ProductForm;
use Livewire\Component;
use Livewire\WithFileUploads;

class ProductCreate extends Component
{
    use WithFileUploads;

    public ProductForm $form;

    public function save(): void
    {
        $product = $this->form->store();
        $this->form->reset();
        $this->dispatch('product-saved', id: $product->id);
        session()->flash('ok', 'محصول ذخیره شد');
    }

    public function render() { return view('livewire.product-create'); }
}
کدblade
<form wire:submit="save">
    <input type="text" wire:model.blur="form.name" placeholder="نام محصول">
    @error('form.name') <span class="err">{{ $message }}</span> @enderror

    <input type="number" wire:model="form.price" placeholder="قیمت (تومان)">
    @error('form.price') <span class="err">{{ $message }}</span> @enderror

    <input type="file" wire:model="form.image" accept="image/*">
    <div wire:loading wire:target="form.image">در حال آپلود…</div>
    @if ($form->image) <img src="{{ $form->image->temporaryUrl() }}" width="120"> @endif

    <button type="submit" wire:loading.attr="disabled">
        <span wire:loading.remove wire:target="save">ذخیره</span>
        <span wire:loading wire:target="save">در حال ذخیره…</span>
    </button>
</form>
  • wire:model.blur: وقتی کاربر از فیلد خارج شد بررسی کن؛ wire:model.live با هر کلید درخواست می‌فرستد (برای جست‌وجو با .debounce.300ms ترکیب کن).
  • فایل در مرحله‌ی اول به‌صورت موقت آپلود می‌شود؛ تا store() را صدا نزنی ذخیره‌ی دائمی نمی‌شود. image را حتماً اعتبارسنجی کن.
  • برای نمایش فایل‌های ذخیره‌شده از php artisan storage:link استفاده کن.

سرعت و تجربه‌ی روان: Computed، Lazy و Loading

Livewire با هر تعامل یک درخواست می‌فرستد؛ پس باید هوشمندانه‌اش کرد.

کدphp
use Livewire\Attributes\Computed;
use Livewire\Attributes\Lazy;
use Livewire\Attributes\Url;
use Livewire\WithPagination;

#[Lazy]                                   // اول صفحه را نشان بده، کامپوننت سنگین را بعداً بارگذاری کن
class ProductList extends Component
{
    use WithPagination;

    #[Url(as: 'q', history: true)]        // جست‌وجو در آدرس می‌ماند (قابل‌اشتراک)
    public string $search = '';

    public function updatedSearch(): void { $this->resetPage(); }   // با عوض‌شدن جست‌وجو، به صفحه‌ی ۱ برگرد

    #[Computed]
    public function products()
    {
        return Product::query()
            ->when($this->search, fn ($q) => $q->where('name', 'like', "%{$this->search}%"))
            ->latest()->paginate(12);
    }

    public function placeholder() { return view('livewire.skeleton'); }   // اسکلت بارگذاری

    public function render() { return view('livewire.product-list'); }
}
کدblade
<input type="search" wire:model.live.debounce.400ms="search" placeholder="جست‌وجو…">

<div wire:loading.class="opacity-50" wire:target="search">
    @foreach ($this->products as $product)
        <x-product-card :product="$product" wire:key="p-{{ $product->id }}" />
    @endforeach
</div>

{{ $this->products->links() }}
  • #[Computed]: نتیجه در همان درخواست کش می‌شود و چند بار صدا زدن در Blade کوئری اضافه نمی‌زند.
  • wire:key را در حلقه‌ها همیشه بگذار؛ بدون آن، Livewire عناصر را اشتباه با هم یکی می‌کند.
  • wire:navigate روی لینک‌ها، ناوبری را بدون بارگذاری کامل صفحه (SPA مانند) می‌کند.
  • wire:poll.30s برای داده‌ی زنده‌ی ساده (تعداد سفارش‌های جدید) خوب است؛ برای زنده‌ی واقعی از Broadcasting استفاده کن.
  • هر چیزی که روی ویژگی‌های public می‌گذاری برای کاربر قابل‌مشاهده و قابل‌تغییر است؛ داده‌ی حساس را public نگذار و شناسه‌ها را در Action دوباره با Policy بررسی کن.
امنیت

هر متد public کامپوننت مثل یک مسیر HTTP است و کاربر می‌تواند آن را صدا بزند. داخل هر Action مجوز را با $this->authorize() بررسی کن، نه فقط دکمه را در UI مخفی کن.

ارتباط کامپوننت‌ها: Event، مودال و اعلان

کدphp
// کامپوننت اول: پس از ذخیره
$this->dispatch('product-saved', id: $product->id);

// کامپوننت دوم: گوش‌دادن
use Livewire\Attributes\On;

#[On('product-saved')]
public function refresh(int $id): void
{
    unset($this->products);          // کش Computed را باطل کن
    $this->dispatch('toast', message: 'محصول اضافه شد');
}
کدblade
{{-- Alpine برای تعامل کوچک سمت مرورگر؛ Livewire فقط وقتی سرور لازم است --}}
<div x-data="{ open: false }" @toast.window="open = true; setTimeout(() => open = false, 2500)">
    <div x-show="open" x-transition class="toast" x-text="$event?.detail?.message"></div>
</div>

قاعده‌ی کلی: اگر تعامل فقط ظاهری است (باز و بسته‌شدن منو، تب)، با Alpine انجام بده و درخواست سرور نفرست. اگر داده یا منطق سرور لازم است، Livewire.

۹

اشتباه‌های رایج

  1. فراموش‌کردن .live و تعجب از اینکه تایپ هم‌زمان کار نمی‌کند.
  2. نداشتن wire:key در حلقه و پرش یا گیج‌شدن ردیف‌ها.
  3. کامپوننت با چند المنت ریشه.
  4. اعتماد به مقدار public بدون authorize/validate (حفره‌ی امنیتی).
  5. ریختن کوئری سنگین در render() بدون eager loading؛ هر تغییر کوچک دوباره اجرا می‌شود.
  6. استفاده از Livewire برای چیزی که با یک فرم Blade ساده حل می‌شد.
  7. بدون wire:key در حلقه‌ها؛ فیلدها هنگام به‌روزرسانی جابه‌جا می‌شوند.
  8. گذاشتن مدل کامل Eloquent یا داده‌ی حساس در ویژگی‌های public.
  9. wire:model.live روی فیلدهای متنی بدون debounce؛ با هر حرف یک درخواست.
★

تمرین‌ها

تمرین ۱
  1. جست‌وجوی زنده‌ی محصول را بساز و فیلتر دسته و بازه‌ی قیمت را به آن اضافه کن (با #[Url]).
تمرین ۲
  1. دکمه‌ی «افزودن به سبد» و نشان تعداد سبد را بساز که با رویداد به‌روز شود.
  2. یک تست Livewire بنویس که تعداد منفی را رد می‌کند.
تمرین ۳
  1. جدول سفارش‌های پنل ادمین را با مرتب‌سازی، جست‌وجو و صفحه‌بندی با Livewire بساز و دسترسی هر اکشن را با Policy کنترل کن.
تمرین ۴
  1. فرم «افزودن محصول» را با Form Object، اعتبارسنجی لحظه‌ای و آپلود تصویر با پیش‌نمایش بساز.
  2. یک لیست محصولات با جست‌وجوی debounce، صفحه‌بندی و #[Url] بساز؛ آدرس را کپی کن و در تب دیگر باز کن تا مطمئن شوی جست‌وجو حفظ می‌شود.
تمرین ۵
  1. دو کامپوننت بساز که با dispatch و #[On] با هم حرف بزنند (فرم و لیست). یک تست با Livewire::test() برای رفتار ذخیره‌ی فرم بنویس.
