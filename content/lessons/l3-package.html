چرا پکیج بسازیم؟

وقتی یک کد را در دو یا سه پروژه کپی می‌کنی (مثلاً توابع فارسی‌سازی، اتصال به درگاه یا یک کامپوننت Blade)، وقتشان رسیده که به یک «پکیج» تبدیل شود: یک‌بار می‌نویسی، با composer require همه‌جا نصب می‌کنی و بهبودها به همه‌ی پروژه‌ها می‌رسد. پکیج‌های عمومی روی GitHub و Packagist هم یکی از بهترین ارزش‌ها برای رزومه و برند شخصی هستند.

Auto-Discovery

بخش extra.laravel باعث می‌شود Service Provider و Facade بعد از نصب خودکار ثبت شوند و کاربر چیزی را دستی تنظیم نکند. نسخه‌های require را بر اساس نسخه‌های واقعاً پشتیبانی‌شده‌ی خودت تنظیم کن (و تست کن).

کد اصلی

کدphp
<?php

namespace Rayvo\PersianTools;

class PersianTools
{
    public function __construct(private array $config = []) {}

    public function toEnglishDigits(string $value): string
    {
        return strtr($value, [
            '۰' => '0', '۱' => '1', '۲' => '2', '۳' => '3', '۴' => '4',
            '۵' => '5', '۶' => '6', '۷' => '7', '۸' => '8', '۹' => '9',
        ]);
    }

    public function toPersianDigits(string|int $value): string
    {
        return strtr((string) $value, [
            '0' => '۰', '1' => '۱', '2' => '۲', '3' => '۳', '4' => '۴',
            '5' => '۵', '6' => '۶', '7' => '۷', '8' => '۸', '9' => '۹',
        ]);
    }

    public function money(int $amount): string
    {
        $unit = $this->config['currency_label'] ?? 'تومان';
        $text = number_format($amount);

        return ($this->config['persian_digits'] ?? true)
            ? $this->toPersianDigits($text).' '.$unit
            : $text.' '.$unit;
    }
}

<?php

namespace Rayvo\PersianTools;

use Illuminate\Support\Facades\Blade;
use Illuminate\Support\ServiceProvider;

class PersianToolsServiceProvider extends ServiceProvider
{
    public function register(): void
    {
        $this->mergeConfigFrom(__DIR__.'/../config/persian-tools.php', 'persian-tools');

        $this->app->singleton(PersianTools::class, fn ($app) => new PersianTools($app['config']['persian-tools']));
    }

    public function boot(): void
    {
        // php artisan vendor:publish --tag=persian-tools-config
        $this->publishes([
            __DIR__.'/../config/persian-tools.php' => config_path('persian-tools.php'),
        ], 'persian-tools-config');

        // @toman(850000)  ←  ۸۵۰,۰۰۰ تومان
        Blade::directive('toman', fn (string $expression) => "<?php echo app(\\Rayvo\\PersianTools\\PersianTools::class)->money((int) ({$expression})); ?>");

        // اگر پکیج روت/view/migration دارد:
        // $this->loadRoutesFrom(__DIR__.'/../routes/web.php');
        // $this->loadViewsFrom(__DIR__.'/../resources/views', 'persian-tools');
        // $this->loadMigrationsFrom(__DIR__.'/../database/migrations');
    }
}

تست پکیج داخل یک پروژه‌ی واقعی (Path Repository)

لازم نیست قبل از انتشار، پکیج را جایی آپلود کنی. در پروژه‌ی لاراولی که داری، پوشه‌ی پکیج را کنار پروژه یا داخل packages/ بگذار و به Composer بگو از مسیر محلی نصبش کند (به‌صورت symlink؛ هر تغییر فوری اثر می‌کند).

کدjson
{
    "repositories": [
        { "type": "path", "url": "packages/rayvo/persian-tools", "options": { "symlink": true } }
    ]
}
کدbash
composer require rayvo/persian-tools:@dev
php artisan vendor:publish --tag=persian-tools-config

use Rayvo\PersianTools\Facades\PersianTools;

انتشار روی GitHub و Packagist

  1. یک Repository عمومی در GitHub بساز و کد را push کن. فایل‌های README.md (نصب، استفاده، مثال) و LICENSE (مثلاً MIT) را حتماً بگذار.
  2. نسخه را با تگ Semantic Versioning بزن: git tag v1.0.0 && git push --tags.
  3. در packagist.org وارد شو (با GitHub) و آدرس ریپو را Submit کن؛ از آن به بعد composer require rayvo/persian-tools برای همه کار می‌کند.
  4. Webhook گیت‌هاب را فعال کن تا نسخه‌های جدید خودکار در Packagist بیایند.
تغییرنسخه
رفع باگ سازگار با قبلPATCH: 1.0.1
قابلیت جدید سازگار با قبلMINOR: 1.1.0
تغییر ناسازگار با قبلMAJOR: 2.0.0
ارزش برند

یک پکیج کوچک اما تمیز با README خوب، تست و CI سبز، در رزومه‌ی تو بهتر از ده پروژه‌ی نصفه می‌نشیند. بعد از درس CI/CD می‌توانی تست‌ها را روی هر push خودکار اجرا کنی.

ServiceProvider حرفه‌ای: کانفیگ، مایگریشن، View و دستور

کدphp
namespace Rayvo\Sms;

use Illuminate\Support\ServiceProvider;

class SmsServiceProvider extends ServiceProvider
{
    public function register(): void
    {
        // مقدار پیش‌فرض کانفیگ؛ کاربر می‌تواند فقط بخشی را override کند
        $this->mergeConfigFrom(__DIR__ . '/../config/sms.php', 'sms');

        $this->app->singleton(SmsManager::class, fn ($app) => new SmsManager($app['config']['sms']));
    }

    public function boot(): void
    {
        $this->loadMigrationsFrom(__DIR__ . '/../database/migrations');
        $this->loadViewsFrom(__DIR__ . '/../resources/views', 'sms');
        $this->loadTranslationsFrom(__DIR__ . '/../lang', 'sms');
        $this->loadRoutesFrom(__DIR__ . '/../routes/web.php');

        if ($this->app->runningInConsole()) {
            $this->publishes([__DIR__ . '/../config/sms.php' => config_path('sms.php')], 'sms-config');
            $this->publishes([__DIR__ . '/../resources/views' => resource_path('views/vendor/sms')], 'sms-views');
            $this->commands([Console\SendTestSms::class]);
        }
    }
}
کدjson
// composer.json: کشف خودکار (کاربر لازم نیست Provider را دستی ثبت کند)
{
  "name": "rayvo/laravel-sms",
  "description": "Simple SMS gateway abstraction for Laravel",
  "type": "library",
  "license": "MIT",
  "require": { "php": "^8.2", "illuminate/support": "^11.0|^12.0" },
  "require-dev": { "orchestra/testbench": "^9.0|^10.0", "phpunit/phpunit": "^11.0" },
  "autoload": { "psr-4": { "Rayvo\\Sms\\": "src/" } },
  "extra": {
    "laravel": {
      "providers": ["Rayvo\\Sms\\SmsServiceProvider"],
      "aliases": { "Sms": "Rayvo\\Sms\\Facades\\Sms" }
    }
  }
}
  • mergeConfigFrom در register، و publishes فقط در حالت کنسول (runningInConsole) تا درخواست‌های وب سنگین نشوند.
  • برای publish یک تگ (sms-config) بده تا کاربر فقط همان را منتشر کند: php artisan vendor:publish --tag=sms-config.
  • نسخه‌های سازگار را در require دقیق بنویس و در CI روی همه‌ی آن‌ها تست کن.

انتشار حرفه‌ای: نسخه‌بندی، CI و مستندات

  1. README خوب: نصب (یک دستور)، تنظیم (.env و config)، مثال کوچک و کامل، سناریوی خطا، و جدول نسخه‌های سازگار PHP/Laravel.
  2. نسخه‌بندی معنایی (SemVer): 1.4.2 = «تغییر شکننده».«قابلیت جدید».«رفع باگ». تغییر ناسازگار فقط با افزایش عدد اول.
  3. CHANGELOG و Tag در Git: git tag v1.0.0 && git push --tags.
  4. CI: تست روی ماتریس PHP و لاراول (همان Workflow درس CI/CD) و pint برای استایل.
  5. لایسنس: فایل LICENSE (MIT رایج‌ترین است)؛ بدون آن دیگران قانوناً نمی‌توانند استفاده کنند.
  6. Packagist: مخزن عمومی GitHub را در packagist.org ثبت کن؛ هر Tag جدید خودکار نسخه‌ی جدید می‌شود. برای پکیج خصوصی (مشتری/شرکت) از Satis یا مخزن خصوصی Composer یا ریپازیتوری vcs استفاده کن.
کدjson
// نصب پکیج خصوصی از Git در پروژه‌ی دیگر
"repositories": [{ "type": "vcs", "url": "git@github.com:rayvo/laravel-sms.git" }],
"require": { "rayvo/laravel-sms": "^1.0" }
هدف پکیج

پکیج خوب یک مشکل را خوب حل می‌کند. اگر قابلیت‌های نامربوط اضافه کنی، نگه‌داری سخت می‌شود. یک پکیج کوچک و مستند، بهتر از یک پکیج بزرگ و رهاشده است.

کدphp
// tests/TestCase.php
namespace Rayvo\Sms\Tests;

use Orchestra\Testbench\TestCase as Orchestra;
use Rayvo\Sms\SmsServiceProvider;

abstract class TestCase extends Orchestra
{
    protected function getPackageProviders($app): array
    {
        return [SmsServiceProvider::class];
    }

    protected function defineEnvironment($app): void
    {
        $app['config']->set('sms.driver', 'log');
    }
}

// tests/SmsTest.php
public function test_sms_manager_is_resolved_and_uses_config(): void
{
    $this->assertInstanceOf(\Rayvo\Sms\SmsManager::class, app(\Rayvo\Sms\SmsManager::class));
    $this->assertSame('log', config('sms.driver'));
}

Testbench یک اپلیکیشن لاراول کوچک برای تست پکیج بالا می‌آورد؛ پس نیازی به یک پروژه‌ی کامل نداری. هر کاری که پکیج روی کانفیگ، مسیر و مایگریشن می‌کند، همین‌جا تست‌پذیر است.

اشتباه‌های رایج

  1. قرار دادن کلید/رمز واقعی در پکیج یا فایل config آن.
  2. نداشتن تست و ناسازگاری ناگهانی با نسخه‌ی جدید لاراول.
  3. وابسته‌کردن بی‌دلیل پکیج به کل laravel/framework به‌جای illuminate/support و اجزای لازم.
  4. تغییر ناسازگار در نسخه‌ی PATCH/MINOR (باید MAJOR باشد).
  5. README ناقص؛ کسی که نتواند در ۲ دقیقه نصب و استفاده کند، پکیج را رها می‌کند.
  6. فراموش‌کردن composer dump-autoload بعد از تغییر ساختار.
  7. ثبت‌کردن چیزهای سنگین در boot بدون شرط runningInConsole.
  8. بدون extra.laravel.providers؛ کاربر باید Provider را دستی ثبت کند.
  9. تغییر ناسازگار بدون افزایش عدد اول نسخه (SemVer).
