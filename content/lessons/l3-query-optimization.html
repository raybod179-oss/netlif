اول اندازه بگیر

بهینه‌سازی بدون اندازه‌گیری حدس است. قبل از تغییر بفهم کدام کوئری‌ها کندند و چند تا هستند.

کدphp
<?php

namespace App\Providers;

use Illuminate\Database\Events\QueryExecuted;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\ServiceProvider;

class AppServiceProvider extends ServiceProvider
{
    public function boot(): void
    {
        // هر کوئری کندتر از ۱۰۰ میلی‌ثانیه را لاگ کن (در توسعه و حتی پروداکشن)
        DB::listen(function (QueryExecuted $query) {
            if ($query->time > 100) {
                Log::warning('Slow query', ['sql' => $query->sql, 'ms' => $query->time]);
            }
        });
    }
}
  • ابزارها: Laravel Debugbar (توسعه)، Telescope، لاگ کوئری کند MySQL (slow_query_log).
  • قاعده‌ی سرانگشتی: یک صفحه‌ی معمولی بیش از ~۲۰ کوئری نزند و هیچ کوئری مرتبط با لیست‌ها در حلقه نباشد.

مرور: N+1 و انتخاب ستون

N+1 را در درس Eloquent پیشرفته دیدی؛ همچنان اولین چیزی است که باید بررسی کنی. چند اصل ساده‌ی دیگر:

کدphp
<?php

use App\Models\Product;

// ❌ همه‌ی ستون‌ها (از جمله ستون‌های متن بلند)
$products = Product::with('category')->get();

// ✅ فقط آنچه صفحه نیاز دارد
$products = Product::select('id', 'title', 'price', 'category_id', 'image')
    ->with('category:id,name')
    ->paginate(12);

// ❌ شمردن با لود کردن همه
$count = Product::where('stock', '>', 0)->get()->count();
// ✅ شمردن در دیتابیس
$count = Product::where('stock', '>', 0)->count();

// ✅ فقط وجود؟ exists سریع‌تر از count
$has = Product::where('stock', '>', 0)->exists();

ایندکس: بزرگ‌ترین اهرم سرعت

بدون ایندکس، دیتابیس برای پیداکردن یک سطر کل جدول را می‌خواند (Full Scan). ایندکس مثل فهرست آخر کتاب است. ستونی که در WHERE، JOIN و ORDER BY زیاد استفاده می‌شود کاندید ایندکس است.

کدphp
<?php

use Illuminate\Database\Migrations\Migration;
use Illuminate\Database\Schema\Blueprint;
use Illuminate\Support\Facades\Schema;

return new class extends Migration
{
    public function up(): void
    {
        Schema::table('orders', function (Blueprint $table) {
            $table->index('status');                          // فیلتر روی وضعیت
            $table->index(['user_id', 'created_at']);         // «سفارش‌های کاربر مرتب‌شده بر اساس تاریخ»
        });

        Schema::table('products', function (Blueprint $table) {
            $table->unique('slug');                           // یکتا + ایندکس
        });
    }

    public function down(): void
    {
        Schema::table('orders', function (Blueprint $table) {
            $table->dropIndex(['status']);
            $table->dropIndex(['user_id', 'created_at']);
        });

        Schema::table('products', function (Blueprint $table) {
            $table->dropUnique(['slug']);
        });
    }
};
  • ترتیب ستون‌ها در ایندکس ترکیبی مهم است: (user_id, created_at) برای جست‌وجوی user_id یا هر دو مفید است، ولی تنها برای created_at کمکی نمی‌کند.
  • foreignId()->constrained() خودش ایندکس می‌سازد.
  • ایندکس بیش‌ازحد نوشتن را کند و حجم را زیاد می‌کند؛ فقط جایی که لازم است.
  • جست‌وجوی LIKE '%متن%' از ایندکس معمولی استفاده نمی‌کند؛ برای جست‌وجوی جدی FULLTEXT یا Meilisearch (Laravel Scout).
کدsql
EXPLAIN SELECT * FROM orders WHERE user_id = 5 ORDER BY created_at DESC LIMIT 10;
-- در ستون key ببین ایندکس استفاده شده؛ type=ALL یعنی Full Scan

کش

کدphp
<?php

use App\Models\Category;
use App\Models\Product;
use Illuminate\Support\Facades\Cache;

// داده‌ای که کم عوض می‌شود: ۱۰ دقیقه کش
$categories = Cache::remember('categories:menu', now()->addMinutes(10), function () {
    return Category::select('id', 'name', 'slug')->orderBy('name')->get();
});

// باطل‌کردن کش وقتی داده عوض شد (مثلاً در Observer دسته)
Cache::forget('categories:menu');

// داده‌ی گران (آمار): کش + قفل برای جلوگیری از «هجوم» هم‌زمان
$stats = Cache::flexible('dashboard:stats', [60, 300], function () {   // تازه برای ۶۰ث، کهنه‌ی قابل‌قبول تا ۳۰۰ث
    return ['orders' => App\Models\Order::count(), 'revenue' => App\Models\Order::sum('total')];
});
دو مسئله‌ی سخت کش

ابطال کش (کی باید پاک شود؟) و کش داده‌ی شخصی (هرگز صفحه یا داده‌ی مخصوص یک کاربر را با کلید مشترک کش نکن، وگرنه داده‌ی او به دیگران نشان داده می‌شود). برای شروع CACHE_STORE=database یا file کافی است؛ با رشد، Redis.

Cache::flexible

flexible در نسخه‌های جدید لاراول اضافه شده است؛ اگر پروژه‌ات قدیمی‌تر بود، از remember استفاده کن.

عملیات دسته‌ای و پردازش حجیم

کدphp
<?php

use App\Models\Product;

// ❌ ۱۰۰۰ کوئری
foreach ($rows as $row) {
    Product::create($row);
}

// ✅ یک کوئری
Product::insert($rows);                                   // بدون رویداد/تایم‌استمپ مدل

// ✅ ساخت یا به‌روزرسانی دسته‌ای (بر اساس کلید یکتا)
Product::upsert($rows, ['sku'], ['price', 'stock']);

// ✅ به‌روزرسانی با یک کوئری
Product::where('category_id', 3)->update(['stock' => 0]);

تنظیمات پروداکشن

کدbash
composer install --no-dev --optimize-autoloader   # بدون پکیج‌های توسعه و با autoload بهینه

خواندن EXPLAIN و ایندکس ترکیبی درست

«ایندکس بگذار» کافی نیست؛ باید بدانی MySQL واقعاً از آن استفاده می‌کند یا نه. ابزارش EXPLAIN است.

کدphp
// خروجی EXPLAIN را از Eloquent بگیر
Order::where('user_id', 5)->where('status', 'paid')->orderByDesc('created_at')->explain()->dump();
کدsql
EXPLAIN SELECT * FROM orders
WHERE user_id = 5 AND status = 'paid'
ORDER BY created_at DESC LIMIT 20;
ستونچه چیزی را نشان می‌دهدنشانه‌ی خوب / بد
typeروش دسترسی به جدولخوب: const، ref، range؛ بد: ALL (اسکن کل جدول)
keyایندکسی که انتخاب شدهNULL یعنی هیچ ایندکسی استفاده نشد
rowsتعداد تقریبی ردیف‌های بررسی‌شدههر چه به نتیجه نزدیک‌تر، بهتر
Extraجزئیات اضافیUsing filesort / Using temporary یعنی مرتب‌سازی یا جدول موقت پرهزینه
کدphp
// مایگریشن: ایندکس ترکیبی برای همان کوئری (ترتیب ستون‌ها مهم است)
Schema::table('orders', function (Blueprint $table) {
    $table->index(['user_id', 'status', 'created_at'], 'orders_user_status_created_idx');
});
  • ترتیب ستون‌ها: ستون‌هایی که با = فیلتر می‌شوند اول، بعد ستون بازه (>، BETWEEN)، بعد ستون مرتب‌سازی.
  • ایندکس (a,b,c) برای فیلتر روی a و a,b کار می‌کند ولی برای فیلتر فقط روی b نه (قانون پیشوند چپ).
  • LIKE '%کلمه%' (درصد اول) از ایندکس استفاده نمی‌کند؛ برای جست‌وجوی متنی از FULLTEXT یا ابزار جست‌وجو (Meilisearch/Scout) استفاده کن.
  • WHERE DATE(created_at) = ... ایندکس را بی‌اثر می‌کند؛ به‌جایش بازه بنویس: whereBetween('created_at', [$start, $end]).
  • هر ایندکس نوشتن را کندتر و فضا را بیشتر می‌کند؛ ایندکس بی‌مصرف نساز.

شمارش و جمع بدون کوئری اضافه

کدphp
// ✗ بد: برای هر دسته یک کوئری COUNT (N+1 شمارشی)
foreach ($categories as $c) { echo $c->products()->count(); }

// ✓ خوب: یک کوئری
$categories = Category::withCount('products')->get();      // products_count
$orders = Order::withSum('items', 'total')->withMax('items', 'price')->get();

// ✓ فقط «وجود دارد؟» → exists نه count
if ($user->orders()->where('status', 'paid')->exists()) { /* ... */ }

// ✓ ستون‌های لازم، نه همه
Product::select('id', 'name', 'price', 'slug')->where('active', true)->paginate(24);

// ✓ زیرکوئری برای آخرین سفارش هر کاربر بدون Join پیچیده
$users = User::addSelect([
    'last_order_at' => Order::select('created_at')
        ->whereColumn('user_id', 'users.id')->latest()->limit(1),
])->get();
محاسبه در دیتابیس، نه در PHP

جمع‌زدن هزاران ردیف با $collection->sum() داده‌ی زیادی را به PHP می‌آورد. sum()، avg() و groupBy را به دیتابیس بسپار؛ برای گزارش‌ها از Query Builder و selectRaw استفاده کن.

داده‌ی بزرگ بدون پر شدن حافظه

روشکجا استفاده کننکته
chunkById(500, fn)به‌روزرسانی/پردازش دسته‌ای رکوردهاامن‌ترین برای تغییر ردیف‌ها در حین پردازش
lazy() / lazyById()خواندن میلیون‌ها ردیف، خروجی گرفتن (Export)Collection تنبل؛ هر بار چند ردیف در حافظه
cursor()پیمایش ساده فقط خواندنییک اتصال را باز نگه می‌دارد؛ با Eager Loading سازگار نیست
کدphp
// ارسال اعلان به همه‌ی کاربران فعال، بدون پر شدن RAM
User::where('active', true)->chunkById(500, function ($users) {
    foreach ($users as $user) {
        $user->notify(new Promo());   // بهتر: Job صف‌دار برای هر دسته
    }
});

// خروجی CSV بزرگ با استریم
return response()->streamDownload(function () {
    $out = fopen('php://output', 'w');
    fputcsv($out, ['id', 'total', 'created_at']);
    Order::select('id', 'total', 'created_at')->lazyById(1000)->each(fn ($o) => fputcsv($out, $o->only('id', 'total', 'created_at')));
    fclose($out);
}, 'orders.csv');

نگهبان‌های خودکار: جلوگیری از N+1 و شناسایی کوئری کند

کدphp
// app/Providers/AppServiceProvider.php
use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Facades\DB;

public function boot(): void
{
    // در توسعه، هر Lazy Loading خطا می‌دهد تا N+1 همان لحظه پیدا شود
    Model::preventLazyLoading(! app()->isProduction());

    // لاگ کوئری‌های کندتر از ۵۰۰ میلی‌ثانیه
    DB::listen(function ($query) {
        if ($query->time > 500) {
            logger()->warning('slow query', ['sql' => $query->sql, 'ms' => $query->time]);
        }
    });
}
  • MySQL Slow Query Log: slow_query_log=1 و long_query_time=1 را روی سرور روشن کن و هفته‌ای یک بار فایل را بخوان.
  • Laravel Pulse کوئری‌ها، Jobها و درخواست‌های کند را در یک داشبورد نشان می‌دهد؛ Debugbar/Telescope را فقط در توسعه نصب کن.
  • قبل از هر «بهینه‌سازی» اندازه بگیر؛ بعدش دوباره اندازه بگیر. بهینه‌سازی بدون عدد، حدس است.

  7. ایندکس‌گذاری روی تک‌تک ستون‌ها به‌جای ایندکس ترکیبی متناسب با کوئری.

تمرین‌ها

تمرین ۱
  1. با DB::listen تعداد کوئری‌های صفحه‌ی اصلی فروشگاه را لاگ کن، N+1 را پیدا و رفع کن و تعداد قبل/بعد را مقایسه کن.
تمرین ۲
  1. روی جدول سفارش‌ها با ۱۰۰ هزار رکورد (با Seeder/Factory) یک کوئری را قبل و بعد از ایندکس با EXPLAIN مقایسه کن.
تمرین ۳
  1. منوی دسته‌ها را کش کن و در Observer، هنگام تغییر دسته، کش را باطل کن.
  2. لیست بزرگ سفارش‌ها را با simplePaginate بساز.
تمرین ۴
  1. یک کوئری کند از پروژه‌ات (لیست سفارش‌ها با فیلتر وضعیت و کاربر) را با explain() بررسی کن، ایندکس ترکیبی درست بساز و type/rows را قبل و بعد مقایسه کن.
تمرین ۵
  1. صفحه‌ی لیست دسته‌ها را با withCount بازنویسی کن و تعداد کوئری‌ها را با Debugbar قبل و بعد بشمار.
  2. Model::preventLazyLoading() را روشن کن و همه‌ی خطاهای پروژه را با with() رفع کن.
