مشکل: کنترلر چاق

در شروع، همه‌چیز را داخل کنترلر می‌نویسیم؛ بعد از چند ماه یک متد ۱۵۰ خطی داریم که نه می‌شود تستش کرد نه تغییرش داد. هدف این درس: هر کد در جای درست خودش بنشیند.

کدphp
<?php

namespace App\Http\Controllers;

use App\Models\Order;
use App\Models\Product;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

// ❌ کنترلر چاق: اعتبارسنجی، منطق کسب‌وکار، دیتابیس، ایمیل، همه با هم
class CheckoutController extends Controller
{
    public function store(Request $request)
    {
        $request->validate(['address' => 'required|string|max:300']);

        $cart = session('cart', []);
        $total = 0;

        DB::beginTransaction();
        foreach ($cart as $id => $qty) {
            $product = Product::find($id);
            if ($product->stock < $qty) {
                DB::rollBack();
                return back()->with('error', 'موجودی کافی نیست');
            }
            $product->decrement('stock', $qty);
            $total += $product->price * $qty;
        }
        $order = Order::create(['user_id' => auth()->id(), 'total' => $total, 'address' => $request->address]);
        DB::commit();

        // ایمیل، پیامک، لاگ... همه همین‌جا
        return redirect()->route('orders.show', $order);
    }
}

لایهمسئولیتنمونه

Repository لازم است؟

در اغلب پروژه‌های لاراول، Eloquent خودش «Repository» است. لایه‌ی Repository اضافه فقط وقتی ارزش دارد که واقعاً چند منبع داده یا نیاز به جایگزینی داشته باشی؛ در غیر این‌صورت کد را پیچیده می‌کند.

Action کلاس و DTO

منطق ثبت سفارش در یک کلاس مستقل می‌رود: قابل تست، قابل‌استفاده‌ی مجدد (از وب، API یا دستور Artisan) و کنترلر ساده می‌ماند. ورودی‌ها را در یک «DTO» ساده می‌ریزیم تا نوع‌ها مشخص باشد.

کدphp
<?php

namespace App\Data;

final readonly class PlaceOrderData
{
    /** @param array<int, int> $items  [product_id => qty] */
    public function __construct(
        public int $userId,
        public string $address,
        public array $items,
    ) {}
}
کدphp
<?php

namespace App\Actions;

use App\Data\PlaceOrderData;
use App\Events\OrderPlaced;
use App\Exceptions\OutOfStockException;
use App\Models\Order;
use App\Models\Product;
use Illuminate\Support\Facades\DB;

final class PlaceOrderAction
{
    public function execute(PlaceOrderData $data): Order
    {
        $order = DB::transaction(function () use ($data) {
            $total = 0;

            foreach ($data->items as $productId => $qty) {
                // قفل ردیف: دو خریدار هم‌زمان موجودی را منفی نکنند
                $product = Product::whereKey($productId)->lockForUpdate()->firstOrFail();

                if ($product->stock < $qty) {
                    throw new OutOfStockException($product);
                }

                $product->decrement('stock', $qty);
                $total += $product->price * $qty;
            }

            return Order::create([
                'user_id' => $data->userId,
                'address' => $data->address,
                'total'   => $total,
                'status'  => 'pending',
            ]);
        });

        OrderPlaced::dispatch($order);   // ایمیل و ... در Listenerها (بعد از commit)

        return $order;
    }
}
کدphp
<?php

namespace App\Exceptions;

use App\Models\Product;
use RuntimeException;

class OutOfStockException extends RuntimeException
{
    public function __construct(public readonly Product $product)
    {
        parent::__construct("موجودی «{$product->title}» کافی نیست.");
    }
}
کدphp
<?php

namespace App\Http\Controllers;

use App\Actions\PlaceOrderAction;
use App\Data\PlaceOrderData;
use App\Exceptions\OutOfStockException;
use App\Http\Requests\CheckoutRequest;
use Illuminate\Http\RedirectResponse;

// ✅ کنترلر لاغر
class CheckoutController extends Controller
{
    public function store(CheckoutRequest $request, PlaceOrderAction $placeOrder): RedirectResponse
    {
        try {
            $order = $placeOrder->execute(new PlaceOrderData(
                userId: $request->user()->id,
                address: $request->validated('address'),
                items: session('cart', []),
            ));
        } catch (OutOfStockException $e) {
            return back()->with('error', $e->getMessage());
        }

        session()->forget('cart');

        return redirect()->route('orders.show', $order);
    }
}

PlaceOrderAction $placeOrder در امضای متد یعنی لاراول (Service Container) خودش شیء را می‌سازد و می‌دهد. در تست می‌توانی نمونه‌ی جعلی بدهی.

کدphp
<?php

namespace App\Providers;

use App\Contracts\ShippingCalculator;
use App\Services\FlatShippingCalculator;
use Illuminate\Support\ServiceProvider;

class AppServiceProvider extends ServiceProvider
{
    public function register(): void
    {
        // هرجا ShippingCalculator خواستند، این را بده؛ فردا می‌توانی عوضش کنی
        $this->app->bind(ShippingCalculator::class, FlatShippingCalculator::class);
    }
}

قوانین نوشتن کد تمیز

  • نام‌گذاری: انگلیسی، دقیق و بدون اختصار: $unpaidOrders نه $o؛ تابع‌ها فعل: calculateTotal().
  • تابع کوچک و تک‌کاره: اگر برای توضیح تابع باید «و» بگویی، دو تابع است.
  • زود برگرد (Early Return): شرط‌های بد را اول رد کن تا تو‌در‌تویی کم شود.
  • عدد و رشته‌ی جادویی نه: از Enum و ثابت استفاده کن (OrderStatus::Paid).
  • تکرار نه (DRY) ولی انتزاع زودهنگام هم نه؛ بار سوم که تکرار شد استخراج کن.
  • کامنت چرا را بگوید، نه چه را؛ کد باید خودش «چه» را بگوید.
کدphp
<?php

// ❌ تو در تو
function discount($user, $total) {
    if ($user) {
        if ($user->isVip()) {
            if ($total > 1000000) {
                return 0.2;
            }
        }
    }
    return 0;
}

// ✅ Early return
function discountRate(?object $user, int $total): float
{
    if (! $user?->isVip()) {
        return 0;
    }

    return $total > 1_000_000 ? 0.2 : 0;
}

فرمت یکدست با Laravel Pint (./vendor/bin/pint) و تحلیل خطا با Larastan (PHPStan) را به پروژه اضافه کن؛ در درس CI/CD آن‌ها را خودکار می‌کنیم.

تست Action (بدون مرورگر)

کدphp
<?php

namespace Tests\Feature;

use App\Actions\PlaceOrderAction;
use App\Data\PlaceOrderData;
use App\Exceptions\OutOfStockException;
use App\Models\Product;
use App\Models\User;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Tests\TestCase;

class PlaceOrderActionTest extends TestCase
{
    use RefreshDatabase;

    public function test_it_creates_order_and_reduces_stock(): void
    {
        $user = User::factory()->create();
        $product = Product::factory()->create(['price' => 100000, 'stock' => 5]);

        $order = app(PlaceOrderAction::class)->execute(
            new PlaceOrderData($user->id, 'تهران', [$product->id => 2])
        );

        $this->assertSame(200000, $order->total);
        $this->assertSame(3, $product->fresh()->stock);
    }

    public function test_it_fails_when_out_of_stock(): void
    {
        $user = User::factory()->create();
        $product = Product::factory()->create(['stock' => 1]);

        $this->expectException(OutOfStockException::class);

        app(PlaceOrderAction::class)->execute(
            new PlaceOrderData($user->id, 'تهران', [$product->id => 3])
        );
    }
}

اشتباه‌های رایج

  1. همه‌چیز را در کنترلر و مدل ریختن (مدل چاق هم مشکل است).
  2. انتزاع بیش‌ازحد: Interface و Repository برای هر کلاس حتی بدون نیاز.
  3. منطق کسب‌وکار داخل Blade یا Migration.
  4. ندادن تراکنش و قفل در عملیات موجودی/پول.
  5. استثناهای بی‌نام (Exception عمومی) به‌جای استثنای دقیق.
  6. refactor بدون تست؛ اول تست، بعد تغییر.

تمرین‌ها

تمرین ۱
  1. یکی از کنترلرهای چاق پروژه‌ات (مثلاً ثبت سفارش) را به Form Request + Action + DTO تبدیل کن.
  2. تست Action را بنویس و مطمئن شو رفتار قبلی حفظ شده.
تمرین ۲
  1. Interface برای ShippingCalculator و دو پیاده‌سازی (ثابت و بر اساس وزن) بساز و در Service Provider یکی را bind کن.
  2. در تست پیاده‌سازی جعلی بده.
تمرین ۳
  1. Pint را نصب و اجرا کن و تغییرات را بررسی کن؛ سپس Enum برای وضعیت سفارش بساز و رشته‌های جادویی را جایگزین کن.
