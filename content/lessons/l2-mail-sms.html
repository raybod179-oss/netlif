ایمیل و پیامک

ایمیل رسید و پیامک کد تایید دو چیزی هستند که تقریباً هر سایت واقعی لازم دارد. اینجا یاد می‌گیری آن‌ها را تمیز، قابل‌تست و قابل‌تعویض بنویسی.

توی این درس چه چیزهایی یاد می‌گیریم؟

  1. تنظیم ایمیل
  2. ساخت Mailable با قالب Markdown
  3. پیامک: Service با Interface
  4. کانال سفارشی برای Notification
  5. کد تایید (OTP) درست
  6. تست بدون ارسال واقعی
  7. اشتباه‌های رایج
  8. تمرین‌ها
۱

تنظیم ایمیل

در توسعه‌ی محلی لازم نیست ایمیل واقعی بفرستی. پیش‌فرض MAIL_MAILER=log است و ایمیل‌ها در storage/logs/laravel.log نوشته می‌شوند. برای دیدن ظاهر ایمیل می‌توانی از سرویس‌های تست مثل Mailtrap یا Mailpit استفاده کنی.

کدtext
# .env (نمونه‌ی SMTP؛ مقدارها را از سرویس‌دهنده بگیر)
MAIL_MAILER=smtp
MAIL_HOST=smtp.example.com
MAIL_PORT=587
MAIL_USERNAME=your-username
MAIL_PASSWORD=your-password
MAIL_FROM_ADDRESS="no-reply@yourdomain.com"
MAIL_FROM_NAME="${APP_NAME}"
رمز در گیت نرود

اطلاعات SMTP فقط در .env باشد؛ آن را commit نکن.

ساخت Mailable با قالب Markdown

کدbash
php artisan make:mail OrderPaid --markdown=mail.orders.paid
کدphp
<?php

namespace App\Mail;

use App\Models\Order;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Mail\Mailable;
use Illuminate\Mail\Mailables\Content;
use Illuminate\Mail\Mailables\Envelope;
use Illuminate\Queue\SerializesModels;

class OrderPaid extends Mailable implements ShouldQueue
{
    use Queueable, SerializesModels;

    public function __construct(public Order $order) {}

    public function envelope(): Envelope
    {
        return new Envelope(subject: 'رسید پرداخت سفارش '.$this->order->number);
    }

    public function content(): Content
    {
        return new Content(
            markdown: 'mail.orders.paid',
            with: ['url' => url('/orders/'.$this->order->id)],
        );
    }
}

ظاهر ایمیل

قالب‌های Markdown را با php artisan vendor:publish --tag=laravel-mail می‌توانی شخصی‌سازی کنی (لوگو، رنگ). ایمیل‌ها در بسیاری از برنامه‌های ایمیل راست‌به‌چپ را خوب نشان نمی‌دهند؛ بعد از ساخت، حتماً در Gmail و موبایل تست کن.

پیامک: Service با Interface

لاراول پیامک داخلی ندارد؛ هر سرویس‌دهنده (کاوه‌نگار، ملی‌پیامک، sms.ir و ...) API خودش را دارد. کار درست این است که یک Interface بسازی و هر سرویس‌دهنده یک پیاده‌سازی باشد؛ اگر روزی سرویس‌دهنده را عوض کنی، فقط یک کلاس عوض می‌شود.

کدphp
<?php

namespace App\Services\Sms;

interface SmsSender
{
    public function send(string $to, string $message): void;
}
کدphp
<?php

namespace App\Services\Sms;

use Illuminate\Support\Facades\Http;
use RuntimeException;

// نمونه: ارسال از طریق یک API وب‌سرویس. آدرس و نام فیلدها را
// طبق مستندات رسمی «سرویس‌دهنده‌ی خودت» تنظیم کن.
class HttpSmsSender implements SmsSender
{
    public function send(string $to, string $message): void
    {
        $response = Http::timeout(10)
            ->retry(2, 200)
            ->acceptJson()
            ->post(config('services.sms.url'), [
                'api_key'  => config('services.sms.key'),
                'receptor' => $to,
                'message'  => $message,
            ]);

        if ($response->failed()) {
            throw new RuntimeException('SMS failed: '.$response->status());
        }
    }
}
کدphp
<?php

namespace App\Services\Sms;

use Illuminate\Support\Facades\Log;

// برای توسعه: به‌جای ارسال واقعی فقط لاگ می‌زند (هزینه‌ی پیامک نمی‌دهی)
class LogSmsSender implements SmsSender
{
    public function send(string $to, string $message): void
    {
        Log::info("SMS to {$to}: {$message}");
    }
}
کدphp
<?php

namespace App\Providers;

use App\Services\Sms\HttpSmsSender;
use App\Services\Sms\LogSmsSender;
use App\Services\Sms\SmsSender;
use Illuminate\Support\ServiceProvider;

class AppServiceProvider extends ServiceProvider
{
    public function register(): void
    {
        // محلی: لاگ؛ پروداکشن: ارسال واقعی
        $this->app->bind(SmsSender::class, fn () => $this->app->isProduction()
            ? new HttpSmsSender()
            : new LogSmsSender());
    }
}

کلید سرویس پیامک را فقط در .env بگذار و از طریق config() بخوان. اگر در گیت برود، دیگران با هزینه‌ی تو پیامک می‌فرستند.

کانال سفارشی برای Notification

کدphp
<?php

namespace App\Notifications\Channels;

use App\Services\Sms\SmsSender;
use Illuminate\Notifications\Notification;

class SmsChannel
{
    public function __construct(private SmsSender $sms) {}

    public function send(object $notifiable, Notification $notification): void
    {
        if (! $notifiable->phone) {
            return;
        }

        $this->sms->send($notifiable->phone, $notification->toSms($notifiable));
    }
}
کدphp
<?php

namespace App\Notifications;

use App\Models\Order;
use App\Notifications\Channels\SmsChannel;
use Illuminate\Notifications\Notification;

class OrderPaidNotification extends Notification
{
    public function __construct(public Order $order) {}

    public function via(object $notifiable): array
    {
        return ['mail', SmsChannel::class];   // ایمیل + پیامک
    }

    public function toSms(object $notifiable): string
    {
        return 'سفارش '.$this->order->number.' پرداخت شد. Rayvo';
    }
}

کد تایید (OTP) درست

  1. کد ۴ تا ۶ رقمی تصادفی با random_int() بساز (نه rand).
  2. کد را هش‌شده یا با انقضای کوتاه (مثلاً ۲ دقیقه) ذخیره کن (جدول یا Cache).
  3. تعداد تلاش و تعداد درخواست ارسال را محدود کن (throttle و شمارنده‌ی تلاش).
  4. بعد از استفاده‌ی موفق، کد را حذف کن.
کدphp
<?php

use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Hash;

$code = (string) random_int(100000, 999999);
Cache::put("otp:{$phone}", Hash::make($code), now()->addMinutes(2));
$sms->send($phone, "کد ورود شما: {$code}");

// تایید
$hash = Cache::get("otp:{$phone}");
if ($hash && Hash::check($request->code, $hash)) {
    Cache::forget("otp:{$phone}");
    // ورود موفق
}

تست بدون ارسال واقعی

کدphp
<?php

namespace Tests\Feature;

use App\Mail\OrderPaid;
use App\Models\Order;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Mail;
use Tests\TestCase;

class MailSmsTest extends TestCase
{
    use RefreshDatabase;

    public function test_mail_content(): void
    {
        Mail::fake();
        $order = Order::factory()->create();

        Mail::to('a@b.com')->send(new OrderPaid($order));

        Mail::assertSent(OrderPaid::class, fn ($m) => $m->hasTo('a@b.com'));
    }

    public function test_http_sms_calls_provider(): void
    {
        config(['services.sms.url' => 'https://sms.test/send', 'services.sms.key' => 'k']);
        Http::fake(['sms.test/*' => Http::response(['ok' => true], 200)]);

        (new \App\Services\Sms\HttpSmsSender())->send('09120000000', 'سلام');

        Http::assertSent(fn ($r) => $r->url() === 'https://sms.test/send');
    }
}

اشتباه‌های رایج

  1. فرستادن ایمیل/پیامک هم‌زمان داخل درخواست کاربر؛ به صف بسپار.
  2. ذخیره‌ی کلید API در کد یا گیت.
  3. OTP بدون انقضا و بدون محدودیت تلاش (قابل حدس‌زدن).
  4. تست با سرویس واقعی و پرداخت هزینه؛ از Mail::fake و Http::fake استفاده کن.
  5. ایمیل‌هایی که به اسپم می‌روند: دامنه‌ی فرستنده را با SPF/DKIM تنظیم کن (با پشتیبانی هاست).
