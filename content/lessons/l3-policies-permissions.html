چرا Authorization؟ (و خطر IDOR)

Authentication یعنی «تو کی هستی؟» (ورود). Authorization یعنی «اجازه‌ی انجام این کار را داری؟». بزرگ‌ترین خطای امنیتی پروژه‌های تازه‌کار همین‌جاست: کاربر ورود کرده و فقط با عوض‌کردن عدد در آدرس، سفارش، فاکتور یا محصول کاربر دیگر را می‌بیند یا ویرایش می‌کند.

کدtext
/orders/15      ← سفارش من
/orders/16      ← سفارش یک نفر دیگر! اگر چک نکنی، باز می‌شود

به این حفره IDOR (دسترسی مستقیم ناامن به شیء) می‌گویند و فقط با چک‌کردن مالکیت یا نقش در سمت سرور بسته می‌شود؛ مخفی‌کردن دکمه در صفحه هیچ امنیتی نمی‌دهد.

۲

Policy: قوانین دسترسی هر مدل

کدbash
php artisan make:policy OrderPolicy --model=Order
کدphp
<?php

namespace App\Policies;

use App\Models\Order;
use App\Models\User;

class OrderPolicy
{
    // قبل از همه‌ی متدها: ادمین همه‌کاره است (null یعنی «بگذار بقیه‌ی متدها تصمیم بگیرند»)
    public function before(User $user, string $ability): ?bool
    {
        return $user->role === 'admin' ? true : null;
    }

    public function viewAny(User $user): bool
    {
        return true;                      // لیست سفارش‌های خودش (در کوئری فیلتر می‌شود)
    }

    public function view(User $user, Order $order): bool
    {
        return $user->id === $order->user_id;
    }

    public function update(User $user, Order $order): bool
    {
        // فقط صاحب سفارش و فقط تا قبل از پرداخت
        return $user->id === $order->user_id && $order->status === 'pending';
    }

    public function delete(User $user, Order $order): bool
    {
        return false;                     // مشتری حق حذف ندارد
    }
}
ثبت خودکار

اگر مدل App\Models\Order و Policy آن App\Policies\OrderPolicy باشد، لاراول خودش آن‌ها را به هم وصل می‌کند و نیازی به ثبت دستی نیست.

استفاده در کنترلر، روت و Blade

کدphp
<?php

namespace App\Http\Controllers;

use App\Models\Order;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Gate;

class OrderController extends Controller
{
    public function index(Request $request)
    {
        // اول فیلتر در کوئری: کاربر فقط سفارش‌های خودش را می‌بیند
        $orders = $request->user()->orders()->latest()->paginate(10);

        return view('orders.index', compact('orders'));
    }

    public function show(Order $order)
    {
        Gate::authorize('view', $order);         // اگر اجازه نداشت، خودکار ۴۰۳

        return view('orders.show', compact('order'));
    }

    public function update(Request $request, Order $order)
    {
        Gate::authorize('update', $order);

        $order->update($request->validate(['note' => ['nullable', 'string', 'max:500']]));

        return back()->with('success', 'ذخیره شد');
    }
}
کدphp
<?php

// روی روت (Middleware)

کدhtml
@can('update', $order)
    <a href="{{ route('orders.edit', $order) }}">ویرایش</a>
@endcan

@cannot('delete', $order)
    <span>امکان حذف وجود ندارد</span>
@endcannot
دو لایه

@can فقط برای نمایش زیباتر دکمه است. محافظت واقعی Gate::authorize یا can: در سمت سرور است؛ حتماً هر دو را بنویس.

۴۰۳ یا ۴۰۴؟

اگر نمی‌خواهی وجود یک رکورد برای غریبه‌ها لو برود (مثلاً سفارش دیگران)، در Policy به‌جای false می‌توانی Response::denyAsNotFound() برگردانی تا کاربر ۴۰۴ ببیند.

Gate: قوانین بدون مدل

کدphp
<?php

namespace App\Providers;

use App\Models\User;
use Illuminate\Support\Facades\Gate;
use Illuminate\Support\ServiceProvider;

class AppServiceProvider extends ServiceProvider
{
    public function boot(): void
    {
        Gate::define('view-admin-panel', fn (User $user) => in_array($user->role, ['admin', 'support'], true));
    }
}

// استفاده:  Gate::authorize('view-admin-panel');   یا   @can('view-admin-panel')

نقش و دسترسی: ساده یا Spatie؟

برای شروع یک ستون role (و ترجیحاً Enum) کافی است: admin، vendor، customer. وقتی نیاز به «دسترسی‌های ریز» پیدا کردی (مثلاً «ویرایش محصول» جدا از «حذف محصول» و ادمینی که نقش سفارشی بسازد)، پکیج spatie/laravel-permission استاندارد جامعه است.

کدbash
composer require spatie/laravel-permission
php artisan vendor:publish --provider="Spatie\Permission\PermissionServiceProvider"
php artisan migrate
کدphp
<?php

namespace App\Models;

use Illuminate\Foundation\Auth\User as Authenticatable;
use Spatie\Permission\Traits\HasRoles;

class User extends Authenticatable
{
    use HasRoles;
}

کدphp
<?php

// bootstrap/app.php  (فقط بخش alias را به فایل خودت اضافه کن)
use Illuminate\Foundation\Application;
use Illuminate\Foundation\Configuration\Exceptions;
use Illuminate\Foundation\Configuration\Middleware;
use Spatie\Permission\Middleware\PermissionMiddleware;
use Spatie\Permission\Middleware\RoleMiddleware;

return Application::configure(basePath: dirname(__DIR__))
    ->withRouting(
        web: __DIR__.'/../routes/web.php',
        api: __DIR__.'/../routes/api.php',
        commands: __DIR__.'/../routes/console.php',
        health: '/up',
    )
    ->withMiddleware(function (Middleware $middleware) {
        $middleware->alias([
            'role'       => RoleMiddleware::class,
            'permission' => PermissionMiddleware::class,
        ]);
    })
    ->withExceptions(function (Exceptions $exceptions) {
        //
    })->create();

// استفاده:  Route::middleware(['auth', 'role:admin'])->prefix('admin')->group(...);
کش نقش‌ها

Spatie دسترسی‌ها را کش می‌کند. بعد از تغییر مستقیم در دیتابیس یا Seeder دستور php artisan permission:cache-reset را بزن. نام دقیق دستورات و تنظیمات را در مستندات پکیج ببین؛ ممکن است با نسخه‌ی لاراولت متفاوت باشد.

تست دسترسی؛ مهم‌ترین تست امنیتی

کدphp
<?php

namespace Tests\Feature;

use App\Models\Order;
use App\Models\User;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Tests\TestCase;

class OrderPolicyTest extends TestCase
{
    use RefreshDatabase;

    public function test_user_cannot_view_someone_elses_order(): void
    {
        $order = Order::factory()->create();            // مالک یک کاربر دیگر
        $stranger = User::factory()->create();

        $this->actingAs($stranger)->get("/orders/{$order->id}")->assertForbidden();
    }

    public function test_owner_can_view_own_order(): void
    {
        $user = User::factory()->create();
        $order = Order::factory()->create(['user_id' => $user->id]);

        $this->actingAs($user)->get("/orders/{$order->id}")->assertOk();
    }

    public function test_admin_can_view_any_order(): void
    {
        $admin = User::factory()->create(['role' => 'admin']);
        $order = Order::factory()->create();

        $this->actingAs($admin)->get("/orders/{$order->id}")->assertOk();
    }
}

گام پروژه: فروشنده فقط محصول خودش

در مارکت‌پلیس چندفروشنده، هر فروشنده فقط به محصولات و سفارش‌های خودش دسترسی دارد.

  1. Policy برای Product بساز: update و delete فقط وقتی $product->vendor_id === $user->vendor_id.
  2. لیست پنل فروشنده را با Product::where('vendor_id', ...) محدود کن.
  3. تست بنویس که فروشنده‌ی B نتواند محصول فروشنده‌ی A را ویرایش کند (۴۰۳).
  4. ادمین در before() همه‌کاره باشد.
۸

اشتباه‌های رایج

  1. چک‌نکردن مالکیت و آسیب‌پذیری IDOR (رایج‌ترین).
  2. اتکا به مخفی‌کردن دکمه در Blade به‌جای authorize سمت سرور.
  3. کنترل فقط روی نمایش، نه روی ثبت/ویرایش/حذف.
  4. نقش را از ورودی فرم/درخواست قبول‌کردن (Mass Assignment)؛ role نباید در $fillable باشد.
  5. ننوشتن تست برای دسترسی‌ها.
  6. نشت داده در API: Resource باید فقط چیزی را برگرداند که کاربر حق دیدنش دارد.
★

تمرین‌ها

تمرین ۱
  1. Policy برای Review بساز: هر کاربر فقط نظر خودش را ویرایش/حذف کند و ادمین هر نظری را.
  2. در Blade دکمه‌ها را با @can نشان بده.
تمرین ۲
  1. مسیر /admin را با Gate یا role:admin ببند و تست بنویس که کاربر عادی ۴۰۳ می‌گیرد.
  2. تلاش کن عمداً یک IDOR بسازی (بدون authorize)، آن را با تست نشان بده و بعد ببند.
تمرین ۳
  1. Spatie را نصب کن، نقش‌های admin، editor و customer و چند Permission بساز و با Seeder به کاربران بده.
